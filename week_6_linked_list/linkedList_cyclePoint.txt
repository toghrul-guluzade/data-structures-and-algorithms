ListNode* detectCycle(ListNode *head)
  {
    struct ListNode *slow = head;
    struct ListNode *fast = head;

    while(fast != NULL && fast -> next != NULL){
        fast = fast -> next -> next;
        slow = slow -> next;
        if(slow == fast){
            slow = head;
            while(slow != fast){
                fast = fast -> next;
                slow = slow -> next;
            }
            return slow;

        }
    } 
    return NULL;


  }
